const ease = x => { x = Math.max(0, Math.min(1, x)); return x * x * (3 - 2 * x); };
const ease5 = x => { x = Math.max(0, Math.min(1, x)); return x * x * x * (x * (x * 6 - 15) + 10); };
const easeIn = x => { x = Math.max(0, Math.min(1, x)); return x * x; };

const mod = (a, n) => ((a % n) + n) % n;

const rot = ([x, y], a) => [x * Math.cos(a) - y * Math.sin(a), x * Math.sin(a) + y * Math.cos(a)];

export function prepR(C) {
  if (PREPR.has(C)) return PREPR.get(C);
  const P = prep(C), D = C.def.runner, beat = P.R.P, OFF = P.R.OFF, fix = D.fix || {};
  const merge = (a, b) => { for (const k in b) { if (b[k] && typeof b[k] === 'object' && !Array.isArray(b[k]) && a[k] && typeof a[k] === 'object') merge(a[k], b[k]); else a[k] = b[k]; } return a; };
  const ph = D.course.phones.map(p0 => merge(structuredClone(p0), fix[p0.id] || {}));
  ph.forEach((p, i) => { p.i = i; p.prev = ph[i - 1] || null; p.next = ph[i + 1] || null; });
  const byId = new Map(ph.map(p => [p.id, p])), tippers = ph.filter(p => p.tipper);
  const blocks = (D.blocks || []).map(b => ({ slam: P.hb(b.slam), wallT: P.hb(b.wallT), row: P.items.find(q => q.who === 'S' && q.text === b.text) || null }));
  const petKeys = new Set(Object.keys(D.petunia || {}).filter(k => k !== 'finale'));
  const out = { P, D, beat, OFF, ph, byId, tippers, blocks, petKeys, last: ph[ph.length - 1], quiet: D.quiet || [] };

}

const shiftOf = (RP, p, s) => { let d = 0; for (const T of RP.tippers) if (T.id < p.id && s.tp && s.tp[T.id] && s.tp[T.id].good === false) d += R.SHIFT; return d; };

// ======== a phone at t: where it is, how it's turned, what's inside, is it awake (pure in (t, s.tp)) ========
// x, y: its centre (world) · nat: the angle of the phone itself (an upright phone = 0; lying after a fall clockwise = −π/2) · body: its
// shape now ('U' upright / 'S' on its side) · cang: the angle its content is drawn at · room (→ room1 by ck while it auto-rotates): which
// room the content is · carry: he's carried rigid if he's in it · awake / lock (0 … 1 the shutter) / dark · hang / falling (a tipper)
export function pose(RP, p, t, s) {
  const B = RP.beat, b = (t - RP.OFF) / B, sh = shiftOf(RP, p, s);
  const o = { p, present: true, x: p.cx + sh, y: p.cy, nat: 0, cang: 0, body: p.kind, room: p.kind, room1: null, ck: 0, carry: false, awake: false, lock: 0, dark: 0,
    hang: false, falling: false, crack: false, wob: 0, off: false, dockT: p.tDock ?? 0 };
  if (p.kind === 'S') o.nat = Math.PI / 2;
  if (p.tipper) return tipperPose(RP, p, t, s, o, sh, b);
  if (p.tArrive != null && t < p.tArrive) { o.present = false; return fin(RP, p, t, o); }
  if (p.tDock != null && t < p.tDock) { const [ax, ay] = p.arriveFrom || [5, -1.5], park = p.join === 'D' ? [0, -1.6] : [1.6, 0], k1 = ease5((t - p.tArrive) / B), kd = ease((t - (p.tDock - 0.5 * B)) / (0.5 * B));
    o.x += park[0] * (1 - kd) + (ax - park[0]) * (1 - k1); o.y += park[1] * (1 - kd) + (ay - park[1]) * (1 - k1); }
  if (p.bob && !(p.hold && t >= p.hold[0] && t <= p.hold[1])) o.y += wave(p.bob, b) - p.bob[0] / 2;
  if (p.breathe && b < p.Th + 1) o.x += wave(p.breathe, b);
  const tp = p.tip || p.standUp;
  if (tp) { const up = !!p.standUp, k0 = up ? 'S' : 'U', k1 = up ? 'U' : 'S', W0 = ROOM[k0], n0 = up ? Math.PI / 2 : 0;
    if (t >= tp.tWarn && t < tp.t) o.wob = Math.sin(Math.PI * (t - tp.tWarn) / (0.5 * B)) * 0.07 * clamp((t - tp.tWarn) / 0.15);
    if (t >= tp.t) { const piv = [o.x + W0.w / 2, o.y - W0.h / 2];
      if (t < tp.tLand) { const a = Math.PI / 2 * easeIn((t - tp.t) / (tp.tLand - tp.t)), r = rot([-W0.w / 2, W0.h / 2], -a);
        o.x = piv[0] + r[0]; o.y = piv[1] + r[1]; o.nat = n0 - a; o.cang = -a; o.body = k0; o.room = k0; o.carry = true; o.swing = a; }
      else { o.x = tp.to[0] + sh; o.y = tp.to[1]; o.body = k1; o.nat = n0 - Math.PI / 2; o.landed = t - tp.tLand;
        const [r0, r1] = tp.rotate;
        if (t < r0) { o.cang = -Math.PI / 2; o.room = k0; o.carry = true; }
        else if (t < r1) { const k = ease((t - r0) / (r1 - r0)); o.cang = -Math.PI / 2 * (1 - k); o.room = k0; o.room1 = k1; o.ck = k; o.carry = true; }
        else o.room = k1; } } }
  return fin(RP, p, t, o);
}
function tipperPose(RP, p, t, s, o, sh, b) {
  const B = RP.beat, T = p.tipper, st = s.tp && s.tp[p.id];
  if (t < p.tArrive) { o.present = false; return fin(RP, p, t, o); }
  const hx = p.hangPose[0] + sh, hy = p.hangPose[1];
  if (!st || t < st.t0) { const k1 = ease5((t - p.tArrive) / B); o.x = hx; o.y = hy + 6 * (1 - k1); o.nat = o.cang = 0.087 * Math.sin(Math.PI * b); o.hang = true; o.room = 'U'; return fin(RP, p, t, o, true); }
  const k = clamp((t - st.t0) / (T.fallBeats * B)), tg = st.good ? T.good : T.wrong, tx = tg.c[0] + sh, ty = tg.c[1];
  if (k < 1) { const e = easeIn(k); o.x = hx + (tx - hx) * e; o.y = hy + (ty - hy) * e; o.nat = o.cang = st.good ? -Math.PI / 2 * ease(k) : 0.087 * Math.sin(Math.PI * b) * (1 - k); o.falling = true; o.room = 'U'; return fin(RP, p, t, o, true); }
  const tl = st.t0 + T.fallBeats * B; o.x = tx; o.y = ty; o.landT = tl; o.dockT = tl; o.landed = t - tl;
  if (st.good) { o.body = 'S'; o.nat = -Math.PI / 2; const r0 = tl + 0.15, r1 = tl + 0.45;
    if (t < r0) { o.cang = -Math.PI / 2; o.room = 'U'; } else if (t < r1) { const kk = ease((t - r0) / (r1 - r0)); o.cang = -Math.PI / 2 * (1 - kk); o.room = 'U'; o.room1 = 'S'; o.ck = kk; } else o.room = 'S'; }
  else { o.body = 'U'; o.room = 'U'; o.crack = true; o.y += wave([1.6, 2, 0], (t - tl) / B) - 0.8; }
  return fin(RP, p, t, o);
}
function fin(RP, p, t, o, asleep = false) {
  const B = RP.beat;
  o.lock = p.tSleep0 == null ? 0 : easeIn((t - p.tSleep0) / (p.tSleep1 - p.tSleep0));
  o.awake = o.present && !asleep && t >= o.dockT && (p.tSleep1 == null || t < p.tSleep1);
  if (p.tLeave != null && t >= p.tLeave) { const k = ease((t - p.tLeave) / (2 * B)); o.x -= 3 * k; o.y -= 6 * k; o.dark = 1; if (t > p.tLeave + 2 * B + 0.4) o.present = false; }
  if (t >= R.DARK && p !== RP.last) { o.dark = Math.max(o.dark, clamp((t - R.DARK) / 0.2)); o.y -= 2 * ease((t - R.DARK) / (2 * B)); o.awake = false; o.off = true; o.lock = Math.min(o.lock, 0.999); }
  o.x += WO[0]; o.y += WO[1]; return o;
}

// is the wall between a (his) and b (side ±1, or 0 = the floor) a door right now? → { A, B } poses, else null
export function door(RP, a, b, t, s, side) {
  if (!a || !b) return null; const A = pose(RP, a, t, s), Bp = pose(RP, b, t, s);
  if (!A.present || !Bp.present || !Bp.awake || A.carry || Bp.carry || Bp.hang || Bp.falling || Math.abs(A.cang) > 1e-6 || Math.abs(Bp.cang) > 1e-6 || A.ck || Bp.ck) return null;
  const ka = ROOM[A.body], kb = ROOM[Bp.body];
  if (side) { if (Math.abs(A.x + side * ka.w / 2 - (Bp.x - side * kb.w / 2)) > 0.05) return null; }
  else if (Math.abs(A.y - ka.h / 2 - (Bp.y + kb.h / 2)) > 0.05 || Math.abs(A.x - Bp.x) > 0.2) return null;
  return { A, B: Bp };
}

function kdims(RP, q, K) {
  const c = q._rk || (q._rk = {}); if (c[K.k]) return c[K.k];

    return { lines, w: Math.min(maxw, Math.max(nat, 1.3, minW)), h: lines.length * S0.LH + 2 * S0.PADY }; };
  const sys = text => ({ lines: [text], w: K.TW, h: S0.SYSH });

  if (q.photo) { if (isPet(RP, q)) { const w = K.k === 'S' ? 2.6 : 3.4; d = { nice: { w, h: w, lines: [] }, gl: null }; }
    else { const n = 1 + RP.P.items.filter(o => o.join === q.id || (q.join != null && o.join === q.join)).length, w = Math.min(S0.PHW * K.PHS, (K.TW - (n - 1) * S0.PHG) / n); d = { nice: { w, h: w * S0.PHH / S0.PHW, lines: [] }, gl: null }; } }

  return (c[K.k] = d);
}

function slotOf(q, t, h) {
  if (t < q.e) return [0, 0];
  if (q.who === 'S') { if (t < q.a) { const f = S0.DOTH + q.gap; return [f * ease((t - q.e) / S0.ANIM), f]; } const f = S0.DOTH + q.gap + (h - S0.DOTH) * ease((t - q.a) / S0.ANIM); return [f, f]; }
  if (q.who === 'C') { const f = (S0.DOTH + q.gap) * (1 - ease((t - q.a) / S0.ANIM)); return [Math.min(f, (S0.DOTH + q.gap) * ease((t - q.e) / S0.ANIM)), f]; }
  const f = h + q.gap; return [f * ease((t - q.a) / S0.ANIM), f];
}
// the visible stack in room K at t: [{ q, x0, x1, y0, y1, … }] newest first (Petunia's three photos are three rows, not one)
export function rstack(C, t, s, K) {
  const RP = prepR(C), { items } = RP.P; let hi = -1;
  for (let lo = 0, up = items.length - 1; lo <= up;) { const m = (lo + up) >> 1; if (items[m].a <= t) { hi = m; lo = m + 1; } else up = m - 1; }
  let top = hi; for (let j = hi + 1; j < items.length && items[j].a < t + 2.5; j++) if (items[j].e <= t) top = j;
  const out = [], joins = []; let acc = 0;
  for (let j = top; j >= 0; j--) { const q = items[j]; if (s && !visible(s, q)) continue;
    if (q.join != null && !isPet(RP, q)) { if (t >= q.a) joins.push(q); continue; }
    const d = dimsR(RP, q, t, s, K), [c, full] = slotOf(q, t, d.h); if (c <= 0) continue;
    const y0 = K.TH0 + acc - (full - c); acc += c; if (y0 > K.HB + 2.5) break;
    const dots = (q.who === 'S' && t < q.a) || q.who === 'C', kk = q.who === 'S' && !dots ? ease((t - q.a) / S0.ANIM) : 1;
    const hh = dots ? S0.DOTH * (q.who === 'C' ? clamp(1 - (t - q.a) / S0.ANIM) : 1) : q.who === 'S' ? S0.DOTH + (d.h - S0.DOTH) * kk : d.h, ww = dots ? S0.DOTW : q.who === 'S' ? S0.DOTW + (d.w - S0.DOTW) * kk : d.w;
    const sysOn = !!(q.sys) && d.mine, grey = q.who === 'X' || sysOn;
    const x0 = q.who === 'S' ? -K.CX : q.who === 'C' ? -K.CX + 0.5 : grey ? -ww / 2 : K.CX - ww;
    out.push({ q, x0, x1: x0 + ww, y0, y1: y0 + hh, hh, ww, dots, mine: d.mine, k: d.k, lines: d.lines, sysOn, grey, pop: full > 0 ? c / full : 1 }); }
  for (const q of joins) { const L = out.find(r => r.q.id === q.join); if (!L) continue; const w = L.ww;
    const x1 = L.x0 - S0.PHG - (q.k - 1) * (w + S0.PHG); out.push({ q, x0: x1 - w, x1, y0: L.y0, y1: L.y1, hh: L.hh, ww: w, dots: false, mine: false, k: 0, grey: false, lines: [], pop: clamp((t - q.a) / S0.ANIM) }); }
  if (s && s.knock) for (const r of out) { const kt = s.knock[r.q.id]; if (kt == null || t < kt) continue;
    r.knockT = kt; r.shake = t < kt + J.SHAKE ? (t - kt) / J.SHAKE : 0; r.hole = t >= kt + J.SHAKE; r.solid = storyOn(C); }
  return out;
}

function jPick(P, s, st, t, j, K) {
  const top = histTopK(s, t, K), taken = new Set(Object.values(s.jolt || {}).map(o => o.id).filter(v => v != null)), row = j.pick && j.pick.startsWith('row:') ? j.pick.slice(4) : null;
  const mine = s.ground && typeof s.onId === 'number' ? s.onId : null;
  const ok = (r, age) => r.q.who === 'S' && !r.dots && !r.grey && !r.q.photo && r.q.text && !r.q.tapped && !r.hole && !taken.has(r.q.id) && r.q.id !== mine && t - r.q.a >= age
    && r.y1 < top - 0.3 && r.y0 > K.BAR + 0.1 && !glitchEv(P, s, r.q, t);
  if (row) { const r = st.find(r => r.q.text === row && ok(r, J.ROWAGE)); if (r) return r; }
  let cand = st.filter(r => ok(r, J.AGE)), best = null, bc = 1e9; if (!cand.length) cand = st.filter(r => ok(r, J.ROWAGE));
  for (const r of cand) { const d = r.y0 - s.ly; if (d < J.ABOVE[0] || r.y1 - s.ly > J.ABOVE[1]) continue; if (d < bc) { bc = d; best = r; } }
  if (best) return best;
  for (const r of cand) { const dx = s.lx < r.x0 ? r.x0 - s.lx : s.lx > r.x1 ? s.lx - r.x1 : 0, d = Math.hypot(dx, (r.y0 + r.y1) / 2 - (s.ly + 0.75)); if (d < bc) { bc = d; best = r; } }
  return best;
}
function jLaunch(j, box, aim, t0, warn, K) {   // (the current Ghost's joltLaunch, in room K)
  const w = box.x1 - box.x0, h = box.y1 - box.y0, cx = (box.x0 + box.x1) / 2, cy = (box.y0 + box.y1) / 2;
  let vx = 0, vy = J.DROP_POP, kind = 'drop';
  if (j.kind === 'throw' && aim && aim[1] - cy < J.POP * J.POP / (2 * J.G) - 0.05) {
    let tau = (J.POP + Math.sqrt(J.POP * J.POP - 2 * J.G * (aim[1] - cy))) / J.G; vy = J.POP;
    const need = Math.max(J.MINWARN - (t0 - warn), Math.abs(aim[0] - cx) / J.VXMAX); if (tau < need) { tau = need; vy = (aim[1] - cy + J.G / 2 * tau * tau) / tau; }
    const touch = (tt, v) => { const ux = (aim[0] - cx) / tt; for (let d = 0; d < tt + 0.05; d += 1 / 120) { const x = cx + ux * d, y = cy + v * d - J.G / 2 * d * d, ly = aim[1] - 0.75;
      if (!(aim[0] + 0.6 < x - w / 2 || aim[0] - 0.6 > x + w / 2 || ly + 1.5 < y - h / 2 || ly >= y + h / 2 - 0.08)) return d; } return 1e9; };
    for (let n = 0; n < 12 && touch(tau, vy) + (t0 - warn) < J.MINWARN; n++) { tau += 0.05; vy = (aim[1] - cy + J.G / 2 * tau * tau) / tau; }
    vx = (aim[0] - cx) / tau; kind = 'throw'; }
  const yEnd = K.BAR + 0.2 - h / 2, tb = (vy + Math.sqrt(vy * vy + 2 * J.G * Math.max(0, cy - yEnd))) / J.G, tx = vx ? (Math.sign(vx) * (K.WALL + w / 2) - cx) / vx : 1e9;
  return { j: j.i, t0, w, h, cx, cy, vx, vy, kind, aim, dir: vx ? Math.sign(vx) : j.i % 2 ? 1 : -1, end: t0 + Math.min(J.LIFE, tb, tx) };
}

}

function pickIn(P, s, plats, t, dir, down, K) {
  let best = null, bc = 1e9; const lx = s.lx, ly = s.ly, top = histTopK(s, t, K);
  for (const p of plats) { if (s.ground && p.id === s.onId) continue; if (p.y > ly + 0.35 || ly - p.y > F.TPY || p.y < K.BAR + 0.4 || p.y > top - 0.8) continue;
    const dx = lx < p.x0 ? p.x0 - lx : lx > p.x1 ? lx - p.x1 : 0, cx = (p.x0 + p.x1) / 2; if (dx > F.TPX) continue;
    const gl = glitchEv(P, s, p.r.q, t) ? 1 : 0; let c;
    if (dir) { if ((cx - lx) * dir < 0.6 && !(dx > 0 && (dir > 0 ? p.x0 > lx : p.x1 < lx))) continue; c = Math.abs(ly - p.y) + 0.12 * dx - 3 * gl; }
    else if (down) { if (p.x1 < lx - 1.2 || p.x0 > lx + 1.2 || p.y > ly - 0.2) continue; c = ly - p.y - 3 * gl; }
    else { if (p.y > ly - 0.2) continue; c = (ly - p.y) + 0.3 * dx - (gl && ly - p.y <= F.GLY ? 12 : 0); }
    if (c < bc) { bc = c; best = p; } }
  return best;
}
// through a door: a bubble in the next phone (side ±1: in reach past the wall; 0: the floor) — his glitchable texts first; null = no bubble
function pickDoor(C, RP, s, t, Bp, nx, ny, side) {
  const K = ROOM[Bp.room], P = RP.P, plats = rplats(rstack(C, t, s, K), K), top = Math.min(histTopK(s, t, K), shutterY(Bp, K) ?? 1e9) - R.CLEAR;
  let best = null, bc = 1e9;
  for (const p of plats) { if (p.y > top || p.y < K.BAR + 0.3) continue; const gl = glitchEv(P, s, p.r.q, t) ? 1 : 0; let c;
    if (side) { const near = side > 0 ? p.x0 + K.CX : K.CX - p.x1; if (near > R.REACH || p.y < ny - R.DOWN || p.y > ny + R.UP) continue; c = Math.abs(p.y - ny) + 0.3 * near - 6 * gl; }
    else c = gl ? -100 - p.y : p.y;   // (the floor: the newest of his texts below, else the lowest bubble)
    if (c < bc) { bc = c; best = p; } }
  return best;
}

// one physics step in room K: ride the bubble, fall, land (one-way, from above), the bar's kicks, the walls
function stepBody(C, RP, s, t, K, plats, byId, locked, log) {
  s.vyIn = s.vy;

    if (p && s.lx >= p.x0 - F.FOOT && s.lx <= p.x1 + F.FOOT) { s.ly = p.y; s.vy = 0; } else { s.ground = false; s.onId = null; s.coyT = t; } }
  if (!s.ground) {
    s.vy = Math.max(-F.FALL, s.vy - F.GRAV / 120); s.ly += s.vy / 120;
    let land = null; for (const p of plats) { if (t < s.dropUntil && (s.dropId === 'all' || p.id === s.dropId)) continue; if (s.lx < p.x0 - F.FOOT || s.lx > p.x1 + F.FOOT) continue;
      const pt = s.pt[p.id] ?? p.y; if (s.prevLy - pt >= -0.03 && s.ly - p.y <= 0 && (!land || p.y > land.y)) land = p; }
    if (land) { s.ly = land.y; s.vy = 0; s.ground = true; s.onId = land.id; s.landAt = t; s.bounceN = 0; if (!s.charge) { s.charge = true; log('charge'); } log('land'); if (typeof land.id === 'string') log('jride', { j: +land.id.slice(1) }); }
    else if (s.ly <= K.BAR) {   // the input bar kicks him back up (higher each time in a row; the last one throws him up into the old history)
      const kn = t - (s.knockAt ?? -99) < J.KNOCKSAFE, n = locked || kn ? 0 : Math.min(K.BOUNCE.length - 1, s.bounceN), last = n === K.BOUNCE.length - 1, h = last ? K.HB - K.BAR + 1 : Math.min(K.BOUNCE[n], K.HB - 1.6 - K.BAR);
      s.ly = K.BAR; s.vy = Math.sqrt(2 * F.GRAV * h); if (!kn) s.bounceN++; s.barAt = t; s.barN = n; if (last) s.flung = t;
      s.react = { name: 'wide', dur: 0.4, at: t }; log('bounce', { n }); } }

// a squish (every death): his frame, then the run stops — in the safe times it just pops him through the next door
function squish(s, t, why, log) { if (s.failT != null) return; s.failT = t; s.why0 = why; s.squishAt = t; s.ground = false; s.vy = 0; s.react = { name: 'shock', dur: 1, at: t }; log('fail', { why }); }
// grace / after a retry: the lock just pops him through into the next awake phone (a free glitch)
function popThrough(C, RP, s, t, log) {

  for (const [n, side] of [[rightOf(ph), 1], [below(ph), 0]]) if (n && door(RP, ph, n, t, s, side)) { nb = n; break; }

}
// land him in phone p (at its lowest bubble with room under its header, or the given target)

// through the wall (or the floor) into the next phone: onto a bubble in reach, else out at his own height falling; false = the wall says no

// the auto-rotate's shove (DESIGN §6A): where the content's swing was throwing him (its second half), capped; then normal physics

function place(s, RP, t) {   // his content point → the world (the engine draws him at s.x, s.y)

}

// ======== fair.mjs's look (def.fair.jolts: the level's own timed hazards on the demo's run) ========
// locks: a phone starts to lock ≥ 3 beats after the door out of it opens · tips: wobble ≥ 1.5 beats · tippers: the GOOD window ≥ 0.9 s
// and hanging ≥ 1.5 s before it · BLOCKED ≥ 1.5 s · jolts: a drop's warning (lead + shake) ≥ MINWARN, and standing where he was at
// the warning, its first touch ≥ MINWARN · quiet: nothing near him moves in the quiet windows · 'decides': with the jolts off the same
// presses must not win when they lose with them on
export function fairRunner(C, base, replay) {
  const RP = prepR(C), P = RP.P, out = [], B = RP.beat, s = base;
  for (const p of RP.ph) { const n = p.next; if (!n || p.tSleep0 == null) continue; const open = n.tipper ? n.tipper.tWord + B : n.tDock; if (p.tSleep0 - open < 3 * B - 0.02) out.push({ t: p.tSleep0, kind: 'jolt', v: +(p.tSleep0 - open).toFixed(2), what: `lock #${p.id}: door to #${n.id} open only ${(p.tSleep0 - open).toFixed(2)} s before` }); }
  for (const p of RP.ph) { const tp = p.tip || p.standUp; if (tp && tp.t - tp.tWarn < 1.5 * B - 0.02) out.push({ t: tp.t, kind: 'jolt', v: +(tp.t - tp.tWarn).toFixed(2), what: `tip #${p.id} wobble` });
    if (p.tipper) { const T = p.tipper; if (T.win[1] - T.win[0] < 0.9) out.push({ t: T.tWord, kind: 'jolt', v: +(T.win[1] - T.win[0]).toFixed(2), what: `tipper #${p.id} window` });
      if (T.win[0] - p.tArrive < 1.5) out.push({ t: T.tWord, kind: 'jolt', v: +(T.win[0] - p.tArrive).toFixed(2), what: `tipper #${p.id} hangs too briefly` }); } }
  for (const b of RP.blocks) if (b.wallT - b.slam < 1.5) out.push({ t: b.wallT, kind: 'jolt', v: +(b.wallT - b.slam).toFixed(2), what: 'BLOCKED' });
  for (const [a, z] of RP.quiet) { const cur = currentAt(RP, a), cur2 = currentAt(RP, z);   // (the phone(s) he's meant to be in then hold still — not a locking one behind)
    for (const p of new Set([cur, cur2])) { let moved = false; for (let tt = a; tt <= z; tt += 0.05) { const o0 = pose(RP, p, a, s), o1 = pose(RP, p, tt, s); if (Math.abs(o0.x - o1.x) + Math.abs(o0.y - o1.y) + Math.abs(o0.cang - o1.cang) > 1e-3) moved = true; }
      if (moved) out.push({ t: a, kind: 'jolt', v: 0, what: `#${p.id} moves in the quiet ${a}–${z}` }); }
    for (const j of P.jolts) if (j.hit > a && j.hit < z) out.push({ t: j.hit, kind: 'jolt', v: 0, what: `jolt ${j.word} in the quiet ${a}–${z}` }); }
  for (const j of P.jolts) { const o = s.jolt && s.jolt[j.i]; if (!o || o.id == null || o.warnT == null) continue; const L = o.L, q = P.items[o.id], what = `J${j.i + 1} "${j.word}" on “${q.text}”`;
    const lead = L && L.kind === 'throw' ? null : j.hit - o.warnT + J.SHAKE;
    let reach = null; if (L) for (let d = 0; d < 3 && reach == null; d += 1 / 120) { const b = joltBox(L, L.t0 + d); if (!b) break; const ax = o.aim[0], ay = o.aim[1] - 0.75; if (!(ax + 0.6 < b.x0 || ax - 0.6 > b.x1 || ay + 1.5 < b.y0 || ay >= b.y1 - 0.08)) reach = L.t0 + d - o.warnT; }
    if (lead != null && lead < J.MINWARN - 1e-3) out.push({ t: j.hit, kind: 'jolt', v: +lead.toFixed(2), what: what + ' (warning)' });
    if (reach != null && reach < J.MINWARN - 1e-3) out.push({ t: j.hit, kind: 'jolt', v: +reach.toFixed(2), what: what + ' (reach)' }); }
  for (const k of s.jlog.filter(e => e.k === 'knock')) { const f = s.jlog.find(e => e.k === 'fail' && e.t >= k.t && e.t < k.t + J.KNOCKSAFE + 2); if (f) out.push({ t: k.t, kind: 'knock', v: +(f.t - k.t).toFixed(2), what: `J${k.j + 1} knock → lost` }); }
  if (replay) { const keep = P.jolts; P.jolts = []; let off; try { off = replay(); } finally { P.jolts = keep; }
    if (off.won && !s.won) out.push({ t: s.t, kind: 'decides', v: 0, what: `jolts on: ${s.won ? 'won' : 'lost'} · off: ${off.won ? 'won' : 'lost'}` }); }
  return out;
}
