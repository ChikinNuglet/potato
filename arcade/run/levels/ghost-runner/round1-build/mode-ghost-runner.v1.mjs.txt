const TWU = 2.5 - 2 * PH.FR - 2 * PH.MG;   // the column inside an upright phone (2.0 h)
function bub(text, minW = 0) { const maxw = Math.max(B.MAXF * TWU, minW), lines = wrap(text, maxw - 2 * B.PADX, B.FS), nat = Math.max(...lines.map(l => textW(l, B.FS))) + 2 * B.PADX;
  return { lines, w: Math.min(maxw, Math.max(nat, 0.4, minW)), h: lines.length * B.LH + 2 * B.PADY }; }
const sysB = text => ({ lines: [text], w: TWU, h: B.SYSH });
// the scroll: where home is at t (course x)
export function scrollX(t) { const { V, T0, TW0, TW1, LOCK } = R; t = Math.min(t, LOCK); if (t <= T0) return 0; if (t <= TW0) return V * (t - T0); if (t <= TW1) return V * (TW0 - T0); return V * (t - T0 - (TW1 - TW0)); }
export const towerOn = t => t >= R.TW0 && t < R.TW1 + 0.31;
export function camY(t) {   // the camera's height (h): 0, except down the tower
  if (t < TOWER_CAM[0][0]) return 0;
  if (t >= R.TW1) return -11.6 * (1 - ease((t - R.TW1) / 0.31));
  let y = 0; for (const [tt, yy] of TOWER_CAM) if (t >= tt) y = y + (yy - y) * ease((t - tt) / 0.31); return y;
}
// a piston on the beat: [A, per, phase] → offset (h) and the cable's tell
function moOf(p, t, beat, OFF) { if (!p.motion) return 0; const [A, per, ph] = p.motion, u = mod((t - OFF) / beat - ph, 2 * per);
  return u < 0.5 ? A * ease(u / 0.5) : u < per ? A : u < per + 0.5 ? A * (1 - ease((u - per) / 0.5)) : 0; }
function cableOf(p, t, beat, OFF) { if (!p.motion) return 0; const [, per, ph] = p.motion, u = mod((t - OFF) / beat - ph, 2 * per);
  return u >= 2 * per - 0.5 || u < 0.5 ? 1 : u >= per - 0.5 && u < per + 0.5 ? -1 : 0; }   // (1 taut: rising / about to · -1 wavy: sinking / about to)
// is a phone lit? (a locked one lights when that person's added; at the end they power off behind him, then ahead)
function litOf(RP, p, t) { if (p.locked && t < p.locked) return false;
  if (p === RP.endP) return true;
  if (t >= 110.71 && p.x0 + p.w < scrollX(t) - R.EDGE + 0.6) return false;
  if (t >= 113.17) { const n = Math.floor((t - 113.17) / RP.beat) + 1, k = RP.darkAhead.indexOf(p.id); if (k >= 0 && k < n) return false; }
  return true; }
// a phone right now: its box (h), what it is, can he go in, is it solid, its tip
export function geom(RP, p, t) {
  const { beat, OFF } = RP, lit = litOf(RP, p, t);
  let kind = p.kind, x0 = p.x0, w = p.w, h = p.h, y0 = p.yb + moOf(p, t, beat, OFF), st = 'up', k = 0, rot = 0, piv = null, fin = null;
  if (p.tip) { const d = p.tip.dir;
    fin = d > 0 ? { x0: p.x0 + p.w, y0: p.yb, w: p.h, h: p.w } : { x0: p.x0 - p.h, y0: p.yb, w: p.h, h: p.w };
    if (t >= p.sw1) { kind = 'S'; ({ x0, y0, w, h } = fin); st = 'flat'; }
    else if (t >= p.sw0) { st = 'swing'; k = (t - p.sw0) / beat; rot = -d * (Math.PI / 2) * k * k; piv = [d > 0 ? p.x0 + p.w : p.x0, p.yb]; }
    else if (t >= p.wob) { st = 'wobble'; k = (t - p.wob) / (p.sw0 - p.wob); rot = -d * 0.07 * Math.sin(Math.min(1, k * 1.4) * Math.PI / 2) * (0.75 + 0.25 * Math.sin(t * 40)); piv = [d > 0 ? p.x0 + p.w : p.x0, p.yb]; } }
  if (p.tipUp) { const yb = p.yb + p.upFrozen; fin = { x0: p.x0, y0: yb, w: p.h, h: p.w };
    if (t >= p.sw1) { kind = 'U'; ({ x0, y0, w, h } = fin); st = 'stood'; }
    else if (t >= p.sw0) { y0 = yb; st = 'swing'; k = (t - p.sw0) / beat; rot = (Math.PI / 2) * k * k; piv = [p.x0, yb]; }
    else if (t >= p.wob) { y0 = yb; st = 'wobble'; k = (t - p.wob) / (p.sw0 - p.wob); rot = 0.07 * Math.sin(Math.min(1, k * 1.4) * Math.PI / 2) * (0.75 + 0.25 * Math.sin(t * 40)); piv = [p.x0, yb]; } }
  const enter = (kind === 'U' || kind === 'X') && lit && st !== 'wobble' && st !== 'swing';
  return { p, id: p.id, kind, x0, y0, w, h, x1: x0 + w, y1: y0 + h, st, k, rot, piv, fin, lit, enter, solid: st !== 'swing', cable: cableOf(p, t, beat, OFF) };
}

export const inner = g => ({ ix0: g.x0 + PH.FR, ix1: g.x1 - PH.FR, cx0: g.x0 + PH.FR + PH.MG, bar: g.y0 + PH.FR + PH.BR, th0: g.y0 + PH.FR + PH.BR + 0.08, hb: g.y1 - PH.FR - PH.HD });

// ======== the thread, laid out in h (one layout; every phone shows it) ========
function dimsOf(RP, q, t, s) { const d = RP.dims[q.id];
  if (d.gl && s && isP(s, q)) { const k = ease((t - s.rowP[q.row.t.toFixed(3)]) / B.MORPH); return { w: d.nice.w + (d.gl.w - d.nice.w) * k, h: d.nice.h + (d.gl.h - d.nice.h) * k, k, mine: true, lines: k > 0.5 ? d.gl.lines : d.nice.lines }; }
  return { w: d.nice.w, h: d.nice.h, k: 0, mine: false, lines: d.nice.lines }; }
function slotOf(q, t, h, gap) {
  if (t < q.e) return [0, 0];
  if (q.who === 'S') { if (t < q.a) { const f = B.DOTH + gap; return [f * ease((t - q.e) / B.ANIM), f]; } const f = B.DOTH + gap + (h - B.DOTH) * ease((t - q.a) / B.ANIM); return [f, f]; }
  if (q.who === 'C') { const f = (B.DOTH + gap) * (1 - ease((t - q.a) / B.ANIM)); return [Math.min(f, (B.DOTH + gap) * ease((t - q.e) / B.ANIM)), f]; }
  const f = h + gap; return [f * ease((t - q.a) / B.ANIM), f];
}
// rows, newest first, in the column's own coordinates (x 0..TWU from the left, y up from the thread's floor)
export function tstack(C, t, s, maxY = 5) {
  const RP = prepR(C), items = RP.P.items; let hi = -1;
  for (let lo = 0, up = items.length - 1; lo <= up;) { const m = (lo + up) >> 1; if (items[m].a <= t) { hi = m; lo = m + 1; } else up = m - 1; }
  let top = hi; for (let j = hi + 1; j < items.length && items[j].a < t + 2.5; j++) if (items[j].e <= t) top = j;
  const out = []; let acc = 0;
  for (let j = top; j >= 0; j--) { const q = items[j]; if (s && !visible(s, q)) continue;
    const d = dimsOf(RP, q, t, s), gap = RP.dims[j].gap, [c, full] = slotOf(q, t, d.h, gap); if (c <= 0) continue;
    const y0 = acc - (full - c); acc += c; if (y0 > maxY) break;
    const dots = (q.who === 'S' && t < q.a) || q.who === 'C', kk = q.who === 'S' && !dots ? ease((t - q.a) / B.ANIM) : 1;
    const hh = dots ? B.DOTH * (q.who === 'C' ? clamp(1 - (t - q.a) / B.ANIM) : 1) : q.who === 'S' ? B.DOTH + (d.h - B.DOTH) * kk : d.h, ww = dots ? B.DOTW : q.who === 'S' ? B.DOTW + (d.w - B.DOTW) * kk : d.w;
    const sysOn = !!q.sys && d.mine, grey = q.who === 'X' || sysOn;
    const x0 = q.who === 'S' ? 0 : q.who === 'C' ? 0.05 : grey ? (TWU - ww) / 2 : TWU - ww;
    out.push({ q, x0, x1: x0 + ww, y0, y1: y0 + hh, ww, hh, dots, grey, mine: d.mine, k: d.k, lines: d.lines, pop: full > 0 ? c / full : 1 }); }
  return out;
}

// ======== one frame of the world: the phones near him, their boxes, their bubbles (as platforms) ========

export function world(C, s, t, span = [-7, 11]) {

    const g = geom(RP, p, t); if (g.x1 < lo || g.x0 > hi) continue;

    if (!g.enter && !(g.lit && g.kind !== 'B' && (g.st === 'wobble' || g.kind === 'S'))) continue;
    // its thread: live — frozen once it's behind him (nothing updates out of sight) · Jess's is a screenshot (83.10)

    if (!g.enter) continue;

    g.I = I; g.ht = ht;

      const key = p.id + ':' + r.q.id, kt = s.knock && s.knock[key];
      if (kt != null && t >= kt + J.SHAKE) continue;
      const pl = { key, pid: p.id, g, r, q: r.q, x0, x1, y: y1, y0: I.th0 + r.y0 }; out.plats.push(pl); out.byKey.set(key, pl); }
  }

}

// ======== the phone's jolts (pure flight) ========
function jLaunch(o, box, t0, P) {
  const w = box.x1 - box.x0, h = box.y1 - box.y0, cx = (box.x0 + box.x1) / 2, cy = (box.y0 + box.y1) / 2; let vx = 0, vy = J.DROP_POP;
  if (o.kind === 'throw' && o.aim) { const dx = o.aim[0] - cx, dy = o.aim[1] - cy; let tau = Math.max(J.MINWARN - (t0 - o.warnT), Math.abs(dx) / J.VXMAX, 0.45);
    vy = (dy + J.G / 2 * tau * tau) / tau; vx = dx / tau; }
  Object.assign(o, { t0, w, h, cx, cy, vx, vy, dir: vx ? Math.sign(vx) : 1 });
}
function jBox(o, t, beat) {
